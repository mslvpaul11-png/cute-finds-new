<script>
const BACKEND_URL =
"https://script.google.com/macros/s/AKfycbyLJovLqOpOTpfJ1DYBAM-OCTePOxct0_nl30a5IodWzzzjgSzzD86dILIYawkdrjqt/exec";

let allProducts = [];
let selectedCategory = "All";

function jsonp(url) {
  return new Promise((resolve,reject)=>{
    const callback = "cuteStore_" + Math.random().toString(36).slice(2);
    const script = document.createElement("script");

    const timer = setTimeout(()=>{
      delete window[callback];
      script.remove();
      reject(new Error("Connection timed out."));
    },15000);

    function cleanup(){
      clearTimeout(timer);
      delete window[callback];
      script.remove();
    }

    window[callback] = data=>{
      cleanup();
      resolve(data);
    };

    script.onerror = ()=>{
      cleanup();
      reject(new Error("Unable to connect to the product server."));
    };

    script.src = url +
      (url.includes("?") ? "&" : "?") +
      "callback=" + callback + "&_=" + Date.now();

    document.body.appendChild(script);
  });
}

function safeUrl(value) {
  try {
    const url = new URL(value);
    return url.protocol === "https:" ? url.href : "";
  } catch(e) {
    return "";
  }
}

function renderProducts(){
  const grid = document.getElementById("products");
  const status = document.getElementById("status");
  const query = document.getElementById("search").value.toLowerCase().trim();

  grid.innerHTML = "";

  const filtered = allProducts.filter(p=>{
    const categoryMatch =
      selectedCategory === "All" || p.category === selectedCategory;

    const searchMatch =
      [p.name,p.category,p.description].join(" ").toLowerCase().includes(query);

    return categoryMatch && searchMatch;
  });

  if(!filtered.length){
    status.textContent = "No products found. Please try another search.";
    return;
  }

  status.textContent = "";

  filtered.forEach(p=>{
    const card = document.createElement("article");
    card.className = "card";

    const img = document.createElement("img");
    img.alt = p.name || "Product";
    img.loading = "lazy";
    img.src = safeUrl(p.imageUrl) || "https://placehold.co/500x500/f2e9e4/80675d?text=Cute+Finds";

    const content = document.createElement("div");
    content.className = "card-content";

    const category = document.createElement("div");
    category.className = "category";
    category.textContent = p.category || "Find";

    const title = document.createElement("h3");
    title.textContent = p.name || "Untitled product";

    const description = document.createElement("p");
    description.className = "description";
    description.textContent = p.description || "";

    const price = document.createElement("div");
    price.className = "price";
    price.textContent = p.price || "";

    const link = document.createElement("a");
    link.className = "shop";
    link.textContent = "Shop Now";
    link.href = safeUrl(p.affiliateUrl) || "#";
    link.target = "_blank";
    link.rel = "noopener noreferrer sponsored";

    if(link.href === "#") link.addEventListener("click",e=>e.preventDefault());

    content.append(category,title,description,price,link);
    card.append(img,content);
    grid.appendChild(card);
  });
}

async function loadProducts(){
  document.getElementById("status").textContent = "Loading your finds...";

  try{
    const data = await jsonp(BACKEND_URL + "?action=products");

    if(!data.success) throw new Error(data.error || "Could not load products.");

    allProducts = data.products || [];
    renderProducts();

  }catch(error){
    document.getElementById("status").textContent =
      "Unable to load products. Please try again later.";
    console.error(error);
  }
}

document.getElementById("filters").addEventListener("click",e=>{
  if(e.target.tagName !== "BUTTON") return;

  document.querySelectorAll(".filters button").forEach(b=>b.classList.remove("active"));
  e.target.classList.add("active");

  selectedCategory = e.target.dataset.category;
  renderProducts();
});

document.getElementById("search").addEventListener("input",renderProducts);
document.getElementById("searchBtn").addEventListener("click",renderProducts);
document.getElementById("year").textContent = new Date().getFullYear();

loadProducts();
</script>
