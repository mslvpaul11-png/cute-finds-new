function jsonp(url) {
  return new Promise((resolve, reject) => {
    const callback = "cuteCallback_" +
      Math.random().toString(36).slice(2);

    const script = document.createElement("script");
    const timer = setTimeout(() => {
      cleanup();
      reject(new Error("Server response timed out."));
    }, 15000);

    function cleanup() {
      clearTimeout(timer);
      delete window[callback];
      script.remove();
    }

    window[callback] = data => {
      cleanup();
      resolve(data);
    };

    script.onerror = () => {
      cleanup();
      reject(new Error("Could not connect to backend."));
    };

    script.src = url +
      (url.includes("?") ? "&" : "?") +
      "callback=" + callback + "&_=" + Date.now();

    document.body.appendChild(script);
  });
}

async function sendRequest(action, product) {
  const user = auth.currentUser;
  if (!user) throw new Error("Please sign in again.");

  const token = await user.getIdToken(true);
  const requestId = "req_" + Date.now() + "_" +
    Math.random().toString(36).slice(2);

  const payload = {
    action,
    requestId,
    token,
    product
  };

  const body = new URLSearchParams();
  body.set("payload", JSON.stringify(payload));

  await fetch(BACKEND_URL, {
    method: "POST",
    mode: "no-cors",
    headers: {
      "Content-Type": "application/x-www-form-urlencoded;charset=UTF-8"
    },
    body
  });

  for (let i = 0; i < 12; i++) {
    await new Promise(resolve => setTimeout(resolve, 1500));

    const result = await jsonp(
      BACKEND_URL + "?action=status&requestId=" +
      encodeURIComponent(requestId)
    );

    if (result.success) return result;
    if (!result.pending) {
      throw new Error(result.error || "Request failed.");
    }
  }

  throw new Error("No confirmation received. Refresh products before trying again.");
}

async function compressImage(file) {
  if (!file) return null;

  if (!file.type.startsWith("image/")) {
    throw new Error("Please choose an image file.");
  }

  const image = new Image();
  const localUrl = URL.createObjectURL(file);

  try {
    await new Promise((resolve, reject) => {
      image.onload = resolve;
      image.onerror = reject;
      image.src = localUrl;
    });

    const max = 1400;
    const scale = Math.min(1, max / Math.max(image.width, image.height));
    const canvas = document.createElement("canvas");

    canvas.width = Math.round(image.width * scale);
    canvas.height = Math.round(image.height * scale);

    canvas.getContext("2d").drawImage(
      image, 0, 0, canvas.width, canvas.height
    );

    const blob = await new Promise(resolve =>
      canvas.toBlob(resolve, "image/jpeg", 0.78)
    );

    if (!blob) throw new Error("Image compression failed.");
    if (blob.size > 2 * 1024 * 1024) {
      throw new Error("Image is too large. Choose a smaller image.");
    }

    const base64 = await new Promise((resolve, reject) => {
      const reader = new FileReader();
      reader.onload = () => resolve(reader.result.split(",")[1]);
      reader.onerror = reject;
      reader.readAsDataURL(blob);
    });

    return {
      imageBase64: base64,
      mimeType: "image/jpeg",
      fileName: file.name.replace(/\.[^.]+$/, "") + ".jpg"
    };
  } finally {
    URL.revokeObjectURL(localUrl);
  }
}

$("productForm").onsubmit = async event => {
  event.preventDefault();

  button.disabled = true;

    message("Preparing product...");

    const imageData = await compressImage(imageFile);

      category: $("category").value,
      description: $("description").value.trim(),
      price: $("price").value.trim(),
      affiliateUrl: $("affiliateUrl").value.trim(),
      active: true,
      ...(imageData || {})
    };

    message("Saving product. Please wait...");

    await sendRequest("save", product);

    message("Product saved successfully!");
    clearForm();
    await loadProducts();

  } catch (error) {
    message("Error: " + error.message);
  } finally {
    button.disabled = false;
  }
};

async function loadProducts() {
  $("products").textContent = "Loading...";

  try {
    const data = await jsonp(
      BACKEND_URL + "?action=products"
    );

    if (!data.success) throw new Error(data.error || "Loading failed.");

    const container = $("products");
    container.innerHTML = "";

    if (!data.products.length) {
      container.textContent = "No products added yet.";
      return;
    }

    data.products.forEach(product => {
      const item = document.createElement("div");
      item.className = "product";

      if (product.imageUrl) {
        const img = document.createElement("img");
        img.src = product.imageUrl;
        img.alt = product.name;
        item.appendChild(img);
      }

      const info = document.createElement("div");
      info.className = "product-info";

      const title = document.createElement("strong");
      title.textContent = product.name;

      const detail = document.createElement("p");
      detail.className = "small";
      detail.textContent = product.category + " | " + product.price;

      info.append(title, detail);

      const edit = document.createElement("button");
      edit.textContent = "Edit";
      edit.onclick = () => editProduct(product);

      const del = document.createElement("button");
      del.textContent = "Delete";
      del.className = "danger";
      del.onclick = () => deleteProduct(product.id);

      item.append(info, edit, del);
      container.appendChild(item);
    });

  } catch (error) {
    $("products").textContent = "Error: " + error.message;
  }
}

}

}
